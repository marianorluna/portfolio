## Features

<FeatureGrid>
  <Feature icon="radio" title="Real time">
    Revit changes → WebSocket (pyRevit spool + relay).
  </Feature>
  <Feature icon="eye" title="3D viewer">
    Speckle Viewer linked to the table and chat.
  </Feature>
  <Feature icon="gauge" title="KPIs">
    Cards, charts (Recharts), and a filterable table.
  </Feature>
  <Feature icon="shield" title="QC">
    Data-quality findings.
  </Feature>
  <Feature icon="sparkles" title="AI">
    Natural-language questions (DeepSeek via OpenRouter) with viewer highlighting.
  </Feature>
  <Feature icon="users" title="Roles">
    `admin` (write / ingest) and `guest` (demo with chat quota).
  </Feature>
</FeatureGrid>

## Stack

<StackGrid>
  <StackItem layer="Backend" tool="FastAPI + uvicorn + WebSocket" />
  <StackItem layer="Database" tool="SQLite + SQLAlchemy 2.0 async" />
  <StackItem layer="Frontend" tool="React 19 + Vite + Tailwind" />
  <StackItem layer="3D viewer" tool="@speckle/viewer" />
  <StackItem layer="Charts" tool="Recharts" />
  <StackItem layer="AI" tool="DeepSeek v3 via OpenRouter" />
  <StackItem layer="Auth" tool="JWT (admin / guest)" />
  <StackItem layer="Revit push" tool="pyRevit spool JSONL → relay → /ws/revit" />
  <StackItem layer="Deploy" tool="Docker Compose + Coolify" />
</StackGrid>

## Roles

<RoleTabs ariaLabel="Access roles">
  <RoleCard role="admin" title="Admin">
    Write and ingest: sync the model, manage data, and use chat without the guest quota. Credentials are not public.
  </RoleCard>

  <RoleCard role="guest" title="Guest">
    Public demo with a chat quota: explore KPIs, viewer, and queries up to the daily per-IP limit. Meant to showcase the product without exposing write access.
  </RoleCard>
</RoleTabs>
